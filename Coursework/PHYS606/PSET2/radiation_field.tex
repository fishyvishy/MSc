    The full hamiltonian for the system is given by 
    \[\HH = \HH_{\text{atom}} + \HH_{\text{int}} + \HH_{\text{field}} \]
    Let $\ketbra{i}{j} = \ket{i} \otimes \bra{j_k}$ be a basis for the system with $\ket{i}$ being eigenstates of the atom and $\ket{j_k}$ representing a field with $j$ photons in mode $k$. Setting $\ket{1}$ to be zero energy, the atomic hamiltonian is given by
    \begin{align*}
        \HH_{\text{atom}}&= \hbar\omega_{12}\ketbra{2}{2} + \hbar\omega_{13}\ketbra{3}{3}
    \end{align*}
    The radiation field has hamiltonian 
    \begin{align*}
        \HH_{\text{field}} &= \sum_{k, \lambda} \hbar\nu_{k,\lambda}\ab(a_{k\lambda}^\dagger a_{k\lambda} + \frac{1}{2})
    \end{align*}
    and the interaction hamiltonian is given by 
    \begin{align*}
        \HH_{\text{int}} &= - \hat d \cdot \hat E = -\sum_{\alpha, \alpha'} \ketbra{\alpha}{\alpha} \hat d \ketbra{\alpha'}{\alpha'} \cdot \sum_{k, \lambda} \sqrt{\frac{\hbar \nu_k}{2\e_0 V}} \left(\hat a_{k\lambda}\hat \e_{k\lambda}e^{i\vec k \cdot \vec r} + \hat a_{k\lambda}^\dagger \e_{k\lambda}^* e^{-i\vec k \cdot \vec r}\right)
    \end{align*}
    By assumption, $\uelem{2}{\hat d}{1} = \uelem{3}{\hat d}{1} = \mu \vu z$. Letting $g_k = \frac{\mu}{\hbar}\sqrt{\hbar \nu_k/2\e_0 V}(\hat \e_k \cdot \hat z) e^{i\vec k \cdot \vec r}$, we get  
    \begin{align*}
        \HH_{\text{int}} &= -\hbar\sum_{k, \lambda} g_k \left(\ketbra{2}{1} + \ketbra{3}{1}\right) \hat a_{k\lambda} + g_k^* \left(\ketbra{1}{2} + \ketbra{1}{3}\right) \hat a_{k\lambda}^\dagger
    \end{align*}
    To study spontaneous emission, we consider states $\ket{2, 0}, \ket{3, 0}$ (atom in excited state, no photons) and $\ket{1, 1_k}$ (atom in ground state after emitting photon in mode $k$). We evaluate the matrix elements as 
    \begin{align*}
        \uelem{2, 0}{\HH}{2, 0} &= \hbar \omega_{12} & \uelem{3, 0}{\HH}{3, 0} &= \hbar \omega_{13}\\
        \uelem{1, 1_k}{\HH}{1, 1_k} &= \hbar \nu_k & \uelem{2, 0}{\HH}{1, 1_k} &= -\hbar g_k \\
        \uelem{3, 0}{\HH}{1, 1_k} &= -\hbar g_k &&
    \end{align*}
    where we ignored the $\hbar\nu_k/2$ term in the field hamiltonian since it adds a constant to the total energy, irrelevant to dynamics. Thus, we get 
    \begin{align*}
        \HH &= \underbrace{\hbar\omega_{12}\ketbra{2}{2} + \hbar\omega_{13}\ketbra{3}{3} + \sum_{k}\hbar \nu_k\ketbra{k}{k}}_{:= \HH_0}  - \sum_{k} \ab(\hbar g_k\ketbra{2}{k} + \hbar g_k\ketbra{3}{k} + \text{h.c.})
    \end{align*}
    where we dropped the sum over $\lambda$ by absorbing it implicitly into the sum over $k$.  Let $\HH_1$ be the interaction part of the hamiltonian, i.e $\HH_1 = \HH - \HH_0$. We move to the interaction picture w.r.t $\HH_0$ by defining $\ket{\psi_I} = \hat U^{-1}\ket{\psi}$ where $\hat U = e^{-i\HH_0 t/\hbar}$. Then, 
    \begin{align*}
        i\hbar\odv{\ket{\psi_I}}{t} &= i\hbar \odv{\hat U^{-1}}{t} \ket{\psi} + i\hbar \hat{U}^{-1}\odv{\ket{\psi}}{t} \\
        &= - \HH_0 \hat U^{-1} \ket{\psi} + \hat U^{-1}(\HH_0 + \HH_1) \ket\psi \\
        &= \hat U^{-1} \HH_1 \ket{\psi}\\
        &= \hat U^{-1} \HH_1 \hat U \ket{\psi_I}
    \end{align*}
    where we used the fact that $\hat U^{-1}$ commutes with $\HH_0$ in the last step. Thus, our effective hamiltonian in the interaction picture is $\HH_{\text{eff}} = \hat U^{-1} \HH_1 \hat U$. We note that since $\HH_0$ is diagonal, we have 
    \begin{align*}
        \hat U^{-1}= e^{i\HH_0 t/\hbar} = e^{i\omega_{12}t}\ketbra{2}{2} + e^{i\omega_{13}t}\ketbra{3}{3} + \sum_k e^{i\nu_k t}\ketbra{k}{k}
    \end{align*}
    Hence, we get
    \begin{align*}
        \HH_{\text{eff}} &= -\hbar \sum_k g_k \ketbra{2}{k} e^{-i(\nu_k - \omega_{12})t} + g_k \ketbra{3}{k} e^{-i(\nu_k - \omega_{13})t}+ \text{h.c.}
    \end{align*}
\end{ans}

First, we bring up the energy of the $\ket k$ by making the unitary transformation $\uket{\overline \psi} = U_1 \ket{\psi}$, with $U_1 = e^{-i \sum_k (\nu_k - \omega_{13})t\ketbra{k}{k}} = \sum_k e^{-i(\nu_k - \omega_{13})t}\ketbra{k}{k}$. The last step holds since $\ket k$ is a basis element so $\sum_k \ketbra k k$ is diagonal. We see that 
    \[i\hbar \odv{\ket{\overline \psi}}{t} = i\hbar \ab(\odv{U_1}{t} \ket{\psi} + U_1 \odv{\ket\psi}{t}) = (\underbrace{i\hbar \odv{U_1}{t} U_1^{\dagger} + U_1 \HH_{\text{eff}}U_1^{\dagger}}_{:= \overline{\HH}_{\text{eff}}})\ket{\overline \psi}\]
    We have 
    \[i\hbar \odv{U_1}{t} U_1^{\dagger} = i\hbar \ab(\odv{}{t} e^{-i\sum_k(\nu_k - \omega_{13})t \ketbra{k}{k}})U_1 = \hbar \sum_k(\nu_k - \omega_{13})\ketbra{k}{k} \]

    \[U_1 \HH_{\text{eff}}U_1^\dagger = -\hbar \sum_k g_k \ketbra{2}{k} e^{-i\omega_{23}t} + g_k \ketbra{3}{k} + \text{h.c.}\]

    \[\tub{\overline \HH}{eff} = \hbar \sum_k(\nu_k - \omega_{13})\ketbra{k}{k} -\hbar \sum_k g_k \ketbra{2}{k} e^{-i\omega_{23}t} + g_k \ketbra{3}{k} + \text{h.c.}\]
    To get rid of the $e^{-i\omega_{23}t}$ dependence, we bring down state $\ket 2$ by $\omega_{23}$ with the unitary transform $\uket{\widetilde{\psi}} = U_2 \uket{\overline{\psi}}$ with $U_2 = e^{i\omega_{23}t \ketbra{2}{2}} = e^{i\omega_{23}t}\ketbra{2}{2}$ since $\ket 2$ is a basis element. Then, similar as before, we find 
    \[\tub{\widetilde \HH}{eff} = -\hbar \omega_{23} \ketbra{2}2 +\hbar \sum_k(\nu_k - \omega_{13})\ketbra{k}{k} -\hbar \sum_k g_k \ketbra{2}{k} + g_k \ketbra{3}{k} + \text{h.c.}\]
    Now, we look at the schrodinger equation, $i\hbar \odv{\uket{\widetilde \psi}}{t} = \tub{\widetilde{\HH}}{eff}$, where $\uket{\widetilde{\psi}} = \tilde c_2(t) \ket  2 + \tilde c_3(t) \ket 3 + \sum_k \tilde c_k(t) \ket k$, to get 
    \begin{align*}
        \dot{\tilde c}_2 &=  i\omega_{23} \tilde c_2 + i\sum_k g_k \tilde c_k  \\
        \dot{\tilde c}_3 &=  i\sum_k g_k \tilde c_k \\ 
        \dot{\tilde c}_k &= i\underbrace{(\omega_{13} - \nu_k)}_{:= \delta_k}\tilde c_k + ig_k^* \tilde c_2 + ig_k^* \tilde c_3
    \end{align*}
    We integrate $\dot{\tilde c}_k$ using the integrating factor $e^{-i\delta_k t}$ to get 
    \[c_k(t) = i g_k^*\int_0^t e^{-i\delta_k(t' - t)}(\tilde c_2(t') + \tilde c_3(t')) dt'\]
    where we imposed the intial condition $\tilde c_k(0) = 0$. We assume $\tilde c_2$ and $\tilde c_3$ vary slowly so that
    \[c_k(t) = i g_k^*(\tilde c_2(t) + \tilde c_3(t))\int_0^t e^{-i\delta_k(t' - t)} dt' = ig_k^*(\tilde c_2(t) + \tilde c_3(t))(\pi \delta(\delta_k) + i \pv \frac1{\delta_k})\]  
    Plugging this into $\tilde c_2$ gives us 
    \begin{align*}
        \dot{\tilde c}_2 &= i\omega_{23} \tilde c_2 - (\tilde c_2 + \tilde c_3)\sum_k |g_k|^2(\pi \delta(\delta_k) + i \pv \frac1{\delta_k})
    \end{align*}
    The $i\pv(1/\delta_k)$ term leads to energy shifts (lamb shifts in this context) which we can ignore. Invoking results form the notes, we evaluate 
    \[\sum_k |g_k|^2\pi \delta(\delta_k) = \frac{\mu^2\omega_{13}^2}{3\pi\e_0\hbar c^3} \approx \frac{\mu^2\omega_{12}^2}{3\pi\e_0\hbar c^3} := \frac{\Gamma}{2}\]
    where was assume $\omega_13 \approx \omega_12$. All together, we get 
    \begin{align*}
        \dot{\tilde c}_2 &= i\omega_{23}\tilde c_2 - (\tilde c_2 + \tilde c_3)\Gamma/2 = (i\omega_{23} - \Gamma/2)\tilde c_2 - (\Gamma/2)\tilde c_3
    \end{align*}
    For $\dot{\tilde c}_3$, we similarly get 
    \begin{align*}
        \dot{\tilde c}_3 &=  -(\tilde c_2 + \tilde c_3)\sum_k |g_k|^2(\pi \delta(\delta_k) + i \pv \frac1{\delta_k})\\ 
        &= -(\Gamma/2)\tilde c_3 - (\Gamma/2)\tilde c_2
    \end{align*}
    So, we get 
    \begin{subequations}
        \label{eqn:c2_c3_eoms}
        \begin{align}
            \dot{\tilde c}_2 &=  (i\omega_{23} - \Gamma/2)\tilde c_2 - (\Gamma/2)\tilde c_3 \\ 
        \dot{\tilde c}_3 &= -(\Gamma/2)\tilde c_3 - (\Gamma/2)\tilde c_2
        \end{align}
    \end{subequations}
\end{ans}

 \[A = \begin{pmatrix}
    i\omega_{23} - \Gamma/2 & -\Gamma/2 \\ 
     -\Gamma/2  &  -\Gamma/2 
 \end{pmatrix}\]
 The time evolution of our amplitues is then 
 \begin{align*}
    \begin{pmatrix}
    \tilde c_2(t) \\ \tilde c_3(t)
    \end{pmatrix} = \begin{pmatrix}
        1 \\ 0
    \end{pmatrix}e^{At}
 \end{align*}
 with the initial conditions $\tilde c_2(0) = 1$ and $\tilde c_3(0) = 0$. We solve this using mathetmatica (see accompanying notebook), to get 
  \begin{align*}
    \Gamma \geq \omega, \qquad |c_2(t)|^2 &= \frac{e^{-t\Gamma}\ab((1 + \cosh(\Omega  t))\Gamma^2 - 2\omega^2)}{2\Omega^2}\\
    |c_3(t)|^2 &= \frac{e^{-t\Gamma}\Gamma^2\ab(\cosh(\Omega t) - 1)}{2\Omega^2} \\[1.5em]
    \Gamma < \omega ,\qquad|c_2(t)|^2 &= \frac{e^{-t\Gamma}\ab((1 + \cos(\Omega t))\Gamma^2 - 2\omega^2)}{2\Omega^2} \\ 
    |c_3(t)|^2 &= \frac{e^{-t\Gamma}\Gamma^2\ab(\cos(\Omega t) - 1)}{2\Omega^2} 
 \end{align*}
 with $\Omega = \sqrt{|\Gamma^2 - \omega^2|}$.

For $\Gamma = 0$, we just get $\tilde c_2(t) = e^{-i\omega_{23t}}$ and $\tilde c_3(t) = 0$, which is just the free evolution of the energy states as $\uket{\tilde 2}$ is $\omega_{23}$ above $\uket{\tilde 3}$ in energy. In this case, the system stays in the atomic states and no photons are emitted. We see that the populations have an expontial decay envelope with decay rate $\Gamma$. Thus, $\Gamma$ indicates how often the excited states spontaneously emit a photon.
